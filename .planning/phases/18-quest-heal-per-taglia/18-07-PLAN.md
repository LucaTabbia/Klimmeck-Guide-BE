export const ITEM_USE_CHARACTER_PROJECTION = { status: 1, assets: 1 } as const;

    effect: 1,
    power: 1,

    recovery: 1,
    relatedLore: 1,
} as const;

export interface ItemUseCharacterSnapshot {
    _id: Types.ObjectId;
    status: {
        title: TitleType;
        xp?: number;
        level?: number;
        currentLifePoints?: number;
        maxLifePoints?: number;
        injuries?: InjuryType[];
        awaitingAid?: unknown;
        potionRecoveryUntil?: unknown;
        unlockedLore?: unknown[];
    };
    assets?: InventorySnapshot | null;
}

// oggetto di catalogo letto lean: ogni campo è da sanificare
export interface ItemUseCatalogSnapshot {
    effect?: unknown;
    power?: unknown;
    canHeal?: unknown;
    recovery?: unknown;
    relatedLore?: unknown;
}

export interface ItemUsePlanInput {
    snapshot: ItemUseCharacterSnapshot;
    itemId: Types.ObjectId;
    item: ItemUseCatalogSnapshot | null;
    now: Date;
}

export interface ItemUseOutcome {
    hpBefore: number;
    hpAfter: number;
    injuriesHealed: InjuryType[];
    quantityLeft: number;
    recoveryUntil: Date;
    loreUnlocked: Types.ObjectId[];
}

export interface ItemUsePlan {
    filter: Record<string, unknown>;
    // tipo letterale: mai un'intersezione di interfacce di dominio negli update
    update: {
        $set: Record<string, unknown>;
        $addToSet?: KnowledgeAddition['$addToSet'];
    };
    outcome: ItemUseOutcome;
}

// millisecondi di ricarica validi (intero sicuro > 0) oppure null
export function recoveryTimeOf(recovery: unknown): number | null;
export function planItemUse(input: ItemUsePlanInput): ItemUsePlan;

Regole dei valori:
- potenza = `power` se numero finito e positivo, altrimenti 0; `hpBefore` = `statusFactsOf(snapshot).currentLifePoints`; `hpAfter` = `lifePointsWithin(hpBefore + potenza, massimo)`;
- ferite: `cureInjuries(snapshot.status.injuries ?? [], Array.isArray(item.canHeal) ? item.canHeal : [])`;
- lore: `relatedLore` (se lista) → `refIdOf` su ogni voce, `null` scartati → `knowledgeUpdateOf(snapshot.status.unlockedLore, ids)`;
- pila: riga `{ direction: 'sell', kind: 'LootItem', itemId, quantity: 1 }`; `quantityLeft` = quantità della pila nell'inventario pianificato, 0 se rimossa;
- ricarica in corso solo se `potionRecoveryUntil` è una `Date` con `getTime() > now.getTime()`; ogni altro valore (null, assente, non data) vale come ricarica finita;
- nessun effetto = `hpAfter === hpBefore` e nessuna ferita guarita.
Filtro: `activeCharacterById(snapshot._id)`, `'status.awaitingAid': null`, `'status.currentLifePoints'` = valore GREZZO dello snapshot oppure `null`, `'status.injuries'` = lista dello snapshot (o `[]`), più `soldQuantityGuards([riga], [guardiaRicarica])` dove la guardia è la disgiunzione (operatore `$or`) fra `'status.potionRecoveryUntil': null` (prende anche il campo assente dei legacy, probe P2) e `'status.potionRecoveryUntil'` minore o uguale a `now` (operatore `$lte`).
