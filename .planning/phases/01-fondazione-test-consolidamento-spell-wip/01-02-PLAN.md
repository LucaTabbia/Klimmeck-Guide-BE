- `status.location`: è **required** (ref POI). In `buildCharacter` default un `new Types.ObjectId()`; in `persistCharacter` accettare un `location` via overrides oppure crearne uno con `persistPoi` se assente.

```typescript
import { Model, Types } from 'mongoose';
import { persistPoi } from './poi.fixture';

export function buildCharacter(overrides: Record<string, any> = {}) {
    return {
        infos: {
            sex: 'male', name: 'Test Hero', race: 'human', pronoun: 'he',
            classType: 'fighter', age: 20, background: 'test', imagePath: null,
        },
        status: {
            xp: 20000,        // virtual maxActiveSpells === 1
            level: 1,
            location: new Types.ObjectId(),  // ref POI (required) — override in persist
            title: 'rookie',
            injuries: [],
            spells: [],
            coins: { gold: 0, silver: 0, copper: 0 },
            currentLifePoints: 100,
            maxLifePoints: 100,
        },
        quests: { /* default minimo conforme a CharacterQuests: leggere il modello */ },
        assets: {
            ownedEquipments: [], ownedItems: [],
            wearedEquipment: {}, activeSpells: [], pet: null,
        },
        ...overrides,
    };
}

export async function persistCharacter(
    model: Model<any>,
    overrides: Record<string, any> = {},
    poiModel?: Model<any>,
) {
    const base = buildCharacter(overrides);
    if (!overrides.status?.location && poiModel) {
        const poi = await persistPoi(poiModel);
        base.status.location = poi._id;
    }
    return model.create(base);
}
```
