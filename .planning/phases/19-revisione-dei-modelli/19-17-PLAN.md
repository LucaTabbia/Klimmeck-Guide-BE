```typescript
export interface LegacySpellLists {

    activeSpells?: readonly { spell?: unknown; usages?: unknown }[] | null;       // assets.activeSpells
}
export interface MergedSpellLists {
    spellStates: { spell: Types.ObjectId; usages: number; recoveries: Date[]; active: boolean }[];
    usageConflicts: number;   // usi diversi fra voce e lista delle equipaggiate: vince la voce
    unknownActive: number;    // equipaggiate di magie non conosciute: nessuna voce
    droppedStates: number;    // stati di magie non conosciute: lasciati fuori
}
export function mergeSpellLists(lists: LegacySpellLists, maxUsagesOf: ReadonlyMap<string, number>): MergedSpellLists;
```
