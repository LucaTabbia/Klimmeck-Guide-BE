import { MIN_LIFE_POINTS } from 'src/quests/outcomes/progression/level-table';

// 3 minuti di marcia per power (D-01); si arriva con almeno 1 HP (D-04)
export const FOOD_RULES = {
    minutesPerPower: 3,
    minLifePoints: MIN_LIFE_POINTS,
} as const;

const MS_PER_MINUTE = 60_000;
const MS_PER_POWER = FOOD_RULES.minutesPerPower * MS_PER_MINUTE;

// la durata è già quella del pianificatore: terreno, pet e ferite compresi (D-01)
export function foodNeededFor(durationMs: number): number {
    return durationMs > 0 ? Math.ceil(durationMs / MS_PER_POWER) : 0;
}

// moltiplicare prima di dividere, per eccesso, senza epsilon (D-03, D-07); coperto ⇒ 0 km
export function uncoveredKmOf(needed: number, covered: number, distanceKm: number): number {
    if (needed <= covered) return 0;
    return Math.ceil(((needed - covered) * distanceKm) / needed);
}

// triangolare: il km scoperto n-esimo costa n HP (D-02)
export function hpLostFor(uncoveredKm: number): number {
    return (uncoveredKm * (uncoveredKm + 1)) / 2;
}

Algoritmo, in quest'ordine:
1. `edible` = stack in input filtrati con `Number.isInteger(power) && power > 0 && Number.isInteger(quantity) && quantity >= 1`, deduplicati per id tenendo la PRIMA occorrenza (`sameId`); `sellCopper` non finito o negativo vale 0.
2. `needed <= 0` ⇒ `{ consumed: [], covered: 0 }`.
3. `total = Σ power × quantity`; `total < needed` ⇒ si consuma tutto: `consumed` = ogni stack di `edible` con la sua quantità, nell'ordine di `edible`; `covered = total` (D-04).
4. `byId` = copia di `edible` ordinata per `itemId.toHexString()` crescente (ordine stabile della DP). `maxPower = max(power)`, `cap = needed + maxPower - 1`.
5. Pacchetti: per ogni stack di `byId`, `useful = Math.min(quantity, Math.ceil(cap / power))`; per `k = 1, 2, 4, …` finché `useful > 0`: `n = Math.min(k, useful)`, pacchetto `{ stack, units: n, power: n * power, sell: n * sellCopper }`, `useful -= n`.
6. DP 0/1: `units = new Float64Array(cap + 1).fill(Infinity)`, `sell = new Float64Array(cap + 1).fill(Infinity)`, `units[0] = 0`, `sell[0] = 0`; `taken = packets.map(() => new Uint8Array(cap + 1))`. Per ogni pacchetto `j` nell'ordine, per `s` da `cap` giù fino a `packet.power`: `u = units[s - p] + packet.units`, `v = sell[s - p] + packet.sell`; se `u < units[s] || (u === units[s] && v < sell[s])` (miglioramento STRETTO: a parità vince il pacchetto venuto prima, cioè l'id più basso) allora `units[s] = u`, `sell[s] = v`, `taken[j][s] = 1`.
7. `covered` = il primo `s` in `[needed, cap]` con `units[s]` finito (esiste per costruzione perché `total >= needed`).
8. Ricostruzione: `s = covered`; per `j` dall'ultimo pacchetto al primo, se `taken[j][s]` somma `packet.units` allo stack del pacchetto e `s -= packet.power`.
9. `consumed` = gli stack di `edible` (ordine dell'INPUT, D-17) con quantità sommata > 0, come `{ itemId, quantity }`.

// piano completo della partenza (D-04): tutto o parte del cibo, il resto scoperto; si parte solo con almeno 1 HP all'arrivo
export function planTravelFood(input: TravelFoodInput): TravelFoodPlan {
    const needed = foodNeededFor(input.durationMs);
    const { consumed, covered } = selectFood(needed, input.stacks);
    const uncoveredKm = uncoveredKmOf(needed, covered, input.distanceKm);
    const hpLost = hpLostFor(uncoveredKm);

        consumed,
        needed,
        covered,
        uncoveredKm,
        hpLost,
        hpAfter,
        arrivesAlive: hpAfter >= FOOD_RULES.minLifePoints,
    };
}
