```typescript
    // regole di Phase 8 fuori dalla transazione; dispensa letta in sessione e consumo nello stesso update della partenza (Phase 14 D-12): un rifiuto non scrive nulla
    async startTravel(characterId: Types.ObjectId, poiId: Types.ObjectId): Promise<void> {
        const plan = await this.planner.planFree(characterId, poiId);
        // rieseguito dal driver sui WriteConflict: la dispensa si rilegge, la partenza ricalcola sul cibo del momento (D-10)
        await this.connection.transaction((session) =>
            this.departWithMeal(characterId, plan, session),
        );
        await this.scheduler.schedule(characterId, plan.travel);
    }

    private async departWithMeal(characterId: Types.ObjectId, plan: FreeTravelPlan, session: ClientSession): Promise<void> {
        const { plan: meal, pantry } = await this.pantry.planFor(characterId, plan.travel, session);
        if (!meal.arrivesAlive) throw await this.starvingOrRace(characterId, plan.location, session);
        const started = await this.characterModel
            .findOneAndUpdate(
                startTravelFilter(characterId, plan.location, mealGuardsOf(meal, pantry.hpRaw)),
                startTravelUpdate(plan.travel, mealSetOf(meal, pantry.assets)),
                { session, projection: { _id: 1 } },
            )
            .lean()
            .exec();
        if (!started) throw await this.unstartedError(characterId, session);
    }

    // in corsa il codice della corsa viene prima della fame (D-11): la fame vale solo se la partenza è ancora possibile
    private async starvingOrRace(characterId: Types.ObjectId, location: Types.ObjectId, session: ClientSession): Promise<Error> {
        const startable = await this.characterModel.exists(startTravelFilter(characterId, location)).session(session).exec();
        return startable ? GameRuleException.travelStarving() : this.unstartedError(characterId, session);
    }

    private async unstartedError(characterId: Types.ObjectId, session: ClientSession): Promise<Error> {
        const fresh = await this.characterModel
            .findOne(activeCharacterById(characterId), UNSTARTED_PROJECTION)
            .session(session)
            .lean<UnstartedSnapshot | null>()
            .exec();
        return classifyUnstarted(fresh);
    }
```
