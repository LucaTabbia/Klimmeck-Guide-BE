       { level: 17, minXp: 35_800, maxLifePoints: 28_900 },
       { level: 18, minXp: 40_250, maxLifePoints: 32_400 },
       { level: 19, minXp: 45_000, maxLifePoints: 36_100 },
       { level: 20, minXp: 50_000, maxLifePoints: 40_000 },
   ] as const;

   export type LevelRow = (typeof LEVEL_TABLE)[number];

   export interface Progression { xp: number; level: number; maxLifePoints: number; currentLifePoints: number; }

   const nonNegative = (value: number | null | undefined): number =>
       typeof value === 'number' && Number.isFinite(value) && value > 0 ? Math.floor(value) : 0;

   export function levelRowFor(xp: number): LevelRow {
       let row: LevelRow = LEVEL_TABLE[0];
       for (const candidate of LEVEL_TABLE) if (xp >= candidate.minXp) row = candidate;
       return row;
   }

   // XP clampata (D-11), livello e massimo dalla tabella, nessuna cura al level up (D-12): solo clamp fra 1 e il nuovo massimo
   export function progressionAfter(
       before: { xp?: number | null; currentLifePoints?: number | null },
       xpGained: number,
       hpAfterOutcome: number,
   ): Progression {
       const xp = Math.min(MAX_XP, nonNegative(before.xp) + nonNegative(xpGained));
       const row = levelRowFor(xp);
       const hp = Math.min(row.maxLifePoints, Math.max(MIN_LIFE_POINTS, Math.floor(hpAfterOutcome)));
       return { xp, level: row.level, maxLifePoints: row.maxLifePoints, currentLifePoints: hp };
   }

   export const TITLE_ORDER: readonly TitleType[] = (Object.keys(TITLE_RANK) as TitleType[]).sort((a, b) => TITLE_RANK[a] - TITLE_RANK[b]);

   // avanza solo con la story del titolo esatto (D-14); legend resta legend; nessun avanzamento per XP
   export function nextTitleAfterStory(current: TitleType, storyTitle: TitleType | null | undefined): TitleType {
       if (!storyTitle || current !== storyTitle) return current;

       return next ?? current;
   }
