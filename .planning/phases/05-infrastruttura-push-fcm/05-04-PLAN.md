   ```typescript
   export interface DeviceRegistration { userId: Types.ObjectId; deviceId: string; token: string; platform: PushPlatform; }
   interface StoredToken { token: string; }

   @Injectable()
   export class PushDeviceRegistry {
       constructor(@InjectModel(PushDevice.name) private readonly model: Model<PushDeviceDocument>) {}

       // restituisce il token precedente della riga (userId, deviceId) o null; il chiamante disiscrive dal topic il token sostituito
       async claim(registration: DeviceRegistration, now: Date): Promise<string | null> {
           try {
               return await this.claimOnce(registration, now);
           } catch (error) {
               // un altro utente ha preso lo stesso token tra la delete e l'upsert: il secondo tentativo lo rimuove (RESEARCH Q4)
               if (!isDuplicateKeyError(error)) throw error;
               return this.claimOnce(registration, now);
           }
       }
       private async claimOnce({ userId, deviceId, token, platform }: DeviceRegistration, now: Date): Promise<string | null> {
           await this.model.deleteMany({ token, $nor: [{ userId, deviceId }] }).exec();
           const previous = await this.model
               .findOneAndUpdate({ userId, deviceId }, { $set: { token, platform, lastSeenAt: now } }, { upsert: true, returnDocument: 'before', projection: { token: 1 } })
               .lean<StoredToken | null>()
               .exec();
           return previous?.token ?? null;
       }
       async unregister(userId: Types.ObjectId, deviceId: string): Promise<string | null> {
           const removed = await this.model.findOneAndDelete({ userId, deviceId }, { projection: { token: 1 } }).lean<StoredToken | null>().exec();
           return removed?.token ?? null;
       }
       tokensOf(userId: Types.ObjectId): Promise<string[]> {
           return this.model.distinct('token', { userId }).exec() as Promise<string[]>;
       }
       async removeTokens(tokens: string[]): Promise<number> {
           if (tokens.length === 0) return 0;
           const result = await this.model.deleteMany({ token: { $in: tokens } }).exec();
           return result.deletedCount;
       }
   }
   ```
   Se Mongoose tipizza male `distinct`, usare `.exec()` con cast documentato. NON aggiungere `timestamps` (lastSeenAt è l'unico istante, D-04).
