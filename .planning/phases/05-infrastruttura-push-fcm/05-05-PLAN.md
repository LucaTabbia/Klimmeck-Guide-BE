   export class PushService {
       private readonly logger = new Logger(PushService.name);
       constructor(
           private readonly registry: PushDeviceRegistry,
           private readonly transport: PushTransport,
           private readonly clock: Clock,
           @InjectModel(User.name) private readonly userModel: Model<UserDocument>,
       ) {}

       // best-effort (D-08): mai un'eccezione verso il chiamante di dominio; chiamare DOPO il commit, mai dentro connection.transaction (Pitfall 7)
       async sendToUser(userId: Types.ObjectId, message: PushMessage): Promise<PushDeliveryCounts> {
           if (!this.transport.enabled) return emptyDelivery();
           try {
               const tokens = await this.registry.tokensOf(userId);
               if (tokens.length === 0) return emptyDelivery();
               const outcomes: TokenDeliveryOutcome[] = [];
               const prunable: string[] = [];
               for (const batch of chunk(tokens, FCM_MULTICAST_LIMIT)) {
                   const batchOutcomes = await this.transport.sendToTokens(batch, message);
                   outcomes.push(...batchOutcomes);
                   prunable.push(...prunableTokens(batchOutcomes)); // regola del testimone per batch
               }
               const pruned = await this.registry.removeTokens(prunable);
               this.reportFailures(message.type, outcomes, prunable, pruned);
               return deliveryCountsOf(outcomes, pruned);
           } catch (error) {
               this.logger.warn(`Push ${message.type} not sent: ${describePushError(error)}`);
               return emptyDelivery();
           }
       }

       async broadcast(message: PushMessage): Promise<PushDeliveryCounts> {
           if (!this.transport.enabled) return emptyDelivery();
           try {
               const outcome = await this.transport.sendToTopic(STREAMER_LIVE_TOPIC, message);
               if (!outcome.delivered) this.logger.warn(`Push ${message.type} to topic ${STREAMER_LIVE_TOPIC} not delivered (${outcome.errorCode ?? TRANSPORT_ERROR_CODE})`);
               return topicDeliveryCounts(outcome);
           } catch (error) {
               this.logger.warn(`Push ${message.type} to topic ${STREAMER_LIVE_TOPIC} not sent: ${describePushError(error)}`);
               return topicDeliveryCounts({ delivered: false });
           }
       }
   }
