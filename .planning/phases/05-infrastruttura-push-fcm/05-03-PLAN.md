   ```typescript
   import type { BaseMessage } from 'firebase-admin/messaging';

   const MILLISECONDS_PER_SECOND = 1000;
   export type FcmBase = Required<Pick<BaseMessage, 'data' | 'android' | 'apns'>>;

   // data-only (D-10): il FE gestisce i tre app state e applica i toggle prima della notifica locale
   export function toFcmBase(message: PushMessage, now: Date): FcmBase {
       const data: Record<string, string> = { type: message.type, title: message.title, body: message.body, sentAt: now.toISOString() };
       // mai undefined: un valore non stringa fa fallire ogni token con messaging/invalid-payload
       if (message.id) data.id = message.id;
       const expiresAtEpochSeconds = Math.floor(now.getTime() / MILLISECONDS_PER_SECOND) + message.ttlSeconds;
       return {
           data,
           android: { priority: 'high', ttl: message.ttlSeconds * MILLISECONDS_PER_SECOND }, // ms: l'SDK converte in "3600s"
           apns: {
               headers: {
                   'apns-push-type': 'background',
                   'apns-priority': '5', // 10 è un errore per il push type background (Apple)
                   'apns-expiration': String(expiresAtEpochSeconds), // epoch assoluto in secondi
               },
               payload: { aps: { contentAvailable: true } }, // serializzato come "content-available": 1
           },
       };
   }
   ```

   ```typescript
   import { cert, getApps, initializeApp } from 'firebase-admin/app';
   import { getMessaging, type Messaging, type MessagingTopicManagementResponse } from 'firebase-admin/messaging';

   export const FCM_APP_NAME = 'klimmeck-push';

   export class FcmPushTransport extends PushTransport {
       readonly enabled = true;
       private readonly logger = new Logger('Push');
       constructor(private readonly messaging: Messaging, private readonly clock: Clock) { super(); }

       // nessuna chiamata di rete: il token OAuth si ottiene al primo invio; una seconda initializeApp con lo stesso nome lancerebbe
       static connect(account: FirebaseServiceAccount, clock: Clock): FcmPushTransport {
           const app = getApps().find((candidate) => candidate.name === FCM_APP_NAME) ?? initializeApp({ credential: cert(account) }, FCM_APP_NAME);
           return new FcmPushTransport(getMessaging(app), clock);
       }

       async sendToTokens(tokens: string[], message: PushMessage): Promise<TokenDeliveryOutcome[]> {
           try {
               // l'await dentro il try cattura anche il throw sincrono (0 o > 500 token)
               const batch = await this.messaging.sendEachForMulticast({ tokens, ...toFcmBase(message, this.clock.now()) });
               return batch.responses.map((response, index) => ({ token: tokens[index], delivered: response.success, errorCode: response.error?.code }));
           } catch (error) {
               const errorCode = fcmErrorCodeOf(error);
               return tokens.map((token) => ({ token, delivered: false, errorCode }));
           }
   ```
