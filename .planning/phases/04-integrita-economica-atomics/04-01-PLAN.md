   ```typescript
   export const COPPER_PER_GOLD = 10_000;

   export interface CoinPurse { gold: number; silver: number; copper: number; }

   // rame totale come nel FE (Coins.toCopper): le denominazioni mancanti valgono 0
   export function toCopper(coins: Partial<CoinPurse> | null | undefined): number {
       return (coins?.gold ?? 0) * COPPER_PER_GOLD + (coins?.silver ?? 0) * COPPER_PER_SILVER + (coins?.copper ?? 0);
   }

   export function fromCopper(copper: number): CoinPurse {
       if (!Number.isSafeInteger(copper) || copper < 0) throw new RangeError(`Importo in rame non valido: ${copper}`);
       return {
           gold: Math.floor(copper / COPPER_PER_GOLD),
           silver: Math.floor((copper % COPPER_PER_GOLD) / COPPER_PER_SILVER),
           copper: copper % COPPER_PER_SILVER,
       };
   }

   export const normalize = (coins: Partial<CoinPurse> | null | undefined): CoinPurse => fromCopper(toCopper(coins));
   export const add = (a: Partial<CoinPurse>, b: Partial<CoinPurse>): CoinPurse => fromCopper(toCopper(a) + toCopper(b));
   export const subtract = (a: Partial<CoinPurse>, b: Partial<CoinPurse>): CoinPurse => fromCopper(toCopper(a) - toCopper(b));
   ```

   ```typescript
   import type { UpdateWithAggregationPipeline } from 'mongoose';
   import { COPPER_PER_GOLD, COPPER_PER_SILVER } from './coins';

   export const TOTAL_COPPER_EXPRESSION = {
       $add: [
           { $multiply: [{ $ifNull: ['$status.coins.gold', 0] }, COPPER_PER_GOLD] },
           { $multiply: [{ $ifNull: ['$status.coins.silver', 0] }, COPPER_PER_SILVER] },
           { $ifNull: ['$status.coins.copper', 0] },
       ],
   };

   export function hasAtLeastCopper(amount: number) {
       return { $expr: { $gte: [TOTAL_COPPER_EXPRESSION, amount] } };
   }

   // $toInt su ogni denominazione: il tipo BSON resta int e un delta zero è un vero no-op (nessun evento del change stream)
   export function coinsDeltaStages(deltaCopper: number): UpdateWithAggregationPipeline {
       if (!Number.isSafeInteger(deltaCopper)) throw new RangeError(`Delta in rame non valido: ${deltaCopper}`);
       const total = { $add: [TOTAL_COPPER_EXPRESSION, deltaCopper] };
       return [{ $set: { 'status.coins': {
           gold: { $toInt: { $floor: { $divide: [total, COPPER_PER_GOLD] } } },
           silver: { $toInt: { $floor: { $divide: [{ $mod: [total, COPPER_PER_GOLD] }, COPPER_PER_SILVER] } } },
           copper: { $toInt: { $mod: [total, COPPER_PER_SILVER] } },
       } } }];
   }
   ```
