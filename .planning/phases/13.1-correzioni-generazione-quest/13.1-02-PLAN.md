export interface LeanEnemyCandidate { _id: Types.ObjectId; name?: string | null; xp?: unknown; locations?: unknown; legendary?: unknown; relatedLore?: Types.ObjectId[] | null; baseWeaknesses?: unknown; energyWeaknesses?: unknown; }
export const stringsOf = (value: unknown): string[] => …; // spostata dal service, identica
export function enemyCandidateOf(enemy: LeanEnemyCandidate, animalLoreIds: ReadonlySet<string>): EnemyCandidate { … animal: relatedLore.some((id) => animalLoreIds.has(id)) … }
// solo i non leggendari marcati (D-04 della Phase 13); la regola sugli animali resta nel pool puro (13.1 D-01)
export async function readEnemyCandidates(enemyModel: Model<EnemyDocument>, loreModel: Model<LoreDocument>): Promise<EnemyCandidate[]> {
    const animalLore = await loreModel.find({ type: LoreType.animal }).select('_id').lean<{ _id: Types.ObjectId }[]>().exec();
    const animalLoreIds = new Set(animalLore.map((lore) => lore._id.toHexString()));
    const enemies = await enemyModel.find({ legendary: false }).select(ENEMY_CANDIDATE_FIELDS).sort({ _id: 1 }).lean<LeanEnemyCandidate[]>().exec();
    return enemies.map((enemy) => enemyCandidateOf(enemy, animalLoreIds));
}
