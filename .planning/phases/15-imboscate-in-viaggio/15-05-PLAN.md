```typescript
nearestByTime(from: LonLat, targets: readonly RouteTarget[], fallbacks: readonly LonLat[] = []): NearestRoute | null {
    if (!this.graph) return null;
    const start = this.startNodeFor(from, fallbacks);
    if (!start) return null;
    const paths = dijkstra.singleSource(this.graph, start, (_edge, attributes) => attributes.weight as number);
    let best: NearestRoute | null = null;
    for (const target of [...targets].sort((a, b) => a.id.localeCompare(b.id))) {
        const route = this.routeAlong(paths, start, from, target.location);
        if (route && (!best || route.baseHours < best.route.baseHours)) best = { id: target.id, route };
    }
    return best;
}
```
