export function drawAmbush(input: AmbushDrawInput): DrawnAmbush | null {
    const { distance } = input.road;
    const threshold = input.fromKm === 0 ? AMBUSH_RULES.minTravelKm : AMBUSH_RULES.minRemainingKm;
    if (input.durationMs <= 0 || distance - input.fromKm < threshold) return null;
    for (let km = Math.floor(input.fromKm) + 1; km < distance; km += 1) {
        const at = new Date(input.startTime.getTime() + Math.floor((km / distance) * input.durationMs));
        if (input.chance.next() < hazardAt(input.hourOf(at))) return { at, km };
    }
    return null;
}
