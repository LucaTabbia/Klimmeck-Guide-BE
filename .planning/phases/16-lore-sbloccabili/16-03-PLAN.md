```typescript
async ofMarker(marker: Types.ObjectId | null, session?: ClientSession): Promise<Types.ObjectId[]> {
    if (marker === null) return [];
    const city = await this.cityModel.collection.findOne<{ relatedLore?: unknown }>(
        { markerLocation: { $in: idForms(marker) } },
        { projection: { relatedLore: 1 }, session },
    );
    const cityLoreId = refIdOf(city?.relatedLore);
    if (cityLoreId === null) return [];
    const cityLore = await this.loreModel
        .findById(cityLoreId, { relatedLore: 1 }, { session })
        .lean<{ _id: Types.ObjectId; relatedLore?: unknown[] } | null>()
        .exec();
    if (cityLore === null) return [];
    const relatedIds = (cityLore.relatedLore ?? []).map(refIdOf).filter((id): id is Types.ObjectId => id !== null);
    const related = relatedIds.length === 0 ? [] : await this.loreModel
        .find({ _id: { $in: relatedIds } }, { type: 1 }, { session })
        .lean<GeographyLore[]>()
        .exec();
    const byHex = new Map(related.map((lore) => [lore._id.toHexString(), lore]));
    return geographyOf(cityLore, relatedIds.map((id) => byHex.get(id.toHexString())).filter((lore): lore is GeographyLore => lore !== undefined));
}
```
