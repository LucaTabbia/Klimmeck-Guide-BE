export interface AuditTargetRule {
    type: string;
    idOf: (args: Record<string, unknown>) => unknown;
    argsOf?: (args: Record<string, unknown>) => Record<string, unknown>; // argomenti salvati al posto di quelli ricevuti
    idOfResult?: (result: unknown) => unknown;                            // bersaglio dal risultato, solo alla riuscita
}
export const AuditTarget = (type: string, idOf: AuditTargetRule['idOf'], options: Pick<AuditTargetRule, 'argsOf' | 'idOfResult'> = {}) => SetMetadata(...);

// src/account/deletion/deleted-account.model.ts
