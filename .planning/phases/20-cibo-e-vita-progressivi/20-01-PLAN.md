Forma della curva (RESEARCH §Q2, «il km scoperto n-esimo si paga quando lo si completa», scoperti gli ultimi), con il km del chiamante arrotondato al metro così la differenza è fra interi:

```typescript
export function travelHungerAt(curve: TravelFoodCurve, km: number): number {
    if (km >= curve.distanceKm) return curve.hpLostTotal;
    const remainingMetres =
        Math.round(curve.distanceKm * METRES_PER_KM) -
        Math.round(Math.max(0, km) * METRES_PER_KM);
    const completed = Math.min(
        curve.uncoveredKm,
        Math.max(0, curve.uncoveredKm - Math.ceil(remainingMetres / METRES_PER_KM)),
    );
    return hpLostFor(completed);
}
```
