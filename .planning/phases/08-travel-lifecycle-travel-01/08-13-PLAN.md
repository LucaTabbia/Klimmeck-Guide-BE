       if (!(await this.poiModel.exists({ _id: poiId }))) throw ResourceNotFoundException.pointOfInterest();
       const before = await this.characterModel
           .findOneAndUpdate(teleportFilter(characterId), teleportUpdate(poiId), { returnDocument: 'before', projection: TELEPORT_PROJECTION })
