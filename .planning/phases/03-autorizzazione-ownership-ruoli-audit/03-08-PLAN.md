   ```typescript
   // PendingQuest non referenzia il personaggio: la proprietà si risolve dal personaggio del bearer (D-10)
   async authorizePendingQuest(identity: AuthIdentity, pendingQuestId: string): Promise<ResourceAccess> {
       if (!isObjectIdHex(pendingQuestId)) throw ResourceNotFoundException.pendingQuest();
       if (identity.role === RoleType.innkeeper) return ResourceAccess.onBehalf;
       if ((await this.readOwnPendingQuestId(identity.userId)) === pendingQuestId) return ResourceAccess.owner;
       if (!(await this.pendingQuestModel.exists({ _id: pendingQuestId }))) throw ResourceNotFoundException.pendingQuest();
       throw AuthorizationException.pendingQuestNotOwned();
   }

   private async readOwnPendingQuestId(userId: string): Promise<string | undefined> {
       const characterId = await this.readCurrentCharacterId(userId);
       if (!characterId) return undefined;
       const character = await this.characterModel.findById(characterId, { 'quests.pendingQuest': 1 }).lean().exec();
       return character?.quests?.pendingQuest ? String(character.quests.pendingQuest) : undefined;
   }
   ```
