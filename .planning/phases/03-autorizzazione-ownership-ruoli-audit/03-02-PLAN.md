   ```typescript
   import { Repeater } from '@graphql-yoga/subscription';

   // il Repeater del PubSub aggancia il listener solo al primo next(): va forzato PRIMA di leggere lo snapshot,
   // così un evento pubblicato durante la lettura arriva dopo lo snapshot invece di perdersi (D-03)
   export async function subscribeWithSnapshot<T>(
       live: AsyncIterable<T>,
       loadSnapshot: () => Promise<T>,
   ): Promise<AsyncIterableIterator<T>> {
       const iterator = live[Symbol.asyncIterator]();
       const firstLive = iterator.next();
       let snapshot: T;
       try {
           snapshot = await loadSnapshot();
       } catch (error) {
           await iterator.return?.();
           throw error;
       }
       return new Repeater<T>(async (push, stop) => {
           void stop.then(() => iterator.return?.());
           await push(snapshot);
           for (let result = await firstLive; !result.done; result = await iterator.next()) {
               await push(result.value);
           }
           stop();
       });
   }
   ```
   Vietato: `Repeater.merge`, async generator (`async function*`), lettura dello snapshot prima di `iterator.next()`.

| Boundary | Description |

| server → client (errori HTTP/WS) | message e extensions attraversano il wire; su WS senza `formatDomainError` |
| PubSub → iteratore della subscription | eventi live e snapshot verso un socket per subscriber |
