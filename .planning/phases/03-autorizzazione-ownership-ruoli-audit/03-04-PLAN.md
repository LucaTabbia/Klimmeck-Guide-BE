   export class CharacterAccessPolicy {
       private readonly logger = new Logger(CharacterAccessPolicy.name);

       constructor(
           @InjectModel(User.name) private readonly userModel: Model<UserDocument>,
           @InjectModel(Character.name) private readonly characterModel: Model<CharacterDocument>,
       ) {}

       authorize(identity: AuthIdentity, resource: OwnedResource, id: string): Promise<ResourceAccess> {
           switch (resource) {
               case OwnedResource.character:
                   return this.authorizeCharacter(identity, id);
           }
       }

       // il JWT non porta il personaggio: una sola lettura di users per operazione (D-08); innkeeper esente (D-09);
       // l'esistenza si controlla solo nel ramo di rifiuto, così 403 e 404 restano distinti (D-11)
       async authorizeCharacter(identity: AuthIdentity, characterId: string): Promise<ResourceAccess> {
           if (!isObjectIdHex(characterId)) throw ResourceNotFoundException.character();
           if ((await this.readCurrentCharacterId(identity.userId)) === characterId) return ResourceAccess.owner;
           if (identity.role === RoleType.innkeeper) return ResourceAccess.onBehalf;
           await this.assertCharacterExists(characterId);
           throw AuthorizationException.notOwner();
       }

       // difesa in profondità per ogni evento live: un errore non deve chiudere lo stream né far trapelare l'evento
       async mayReceiveCharacterEvent(identity: AuthIdentity, characterId: string): Promise<boolean> {
           if (identity.role === RoleType.innkeeper) return true;
           try {
               return (await this.readCurrentCharacterId(identity.userId)) === characterId;
           } catch (error) {
               this.logger.warn(`Verifica di proprietà dell'evento fallita: ${error instanceof Error ? error.message : String(error)}`);
               return false;
           }
       }

       private async readCurrentCharacterId(userId: string): Promise<string | undefined> {
           const user = await this.userModel.findById(userId, { currentCharacter: 1 }).lean().exec();
           return user?.currentCharacter ? String(user.currentCharacter) : undefined;
       }

       private async assertCharacterExists(characterId: string): Promise<void> {
           if (!(await this.characterModel.exists({ _id: characterId }))) throw ResourceNotFoundException.character();
       }
   }

   ```typescript
   @Injectable()
   export class OwnershipGuard implements CanActivate {
       constructor(
           private readonly reflector: Reflector,
           private readonly policy: CharacterAccessPolicy,
           private readonly decisions: AuthorizationDecisions,
       ) {}

       async canActivate(context: ExecutionContext): Promise<boolean> {
           const rule = this.reflector.get<OwnershipRule | undefined>(OWNERSHIP_KEY, context.getHandler());
           if (!rule) return true;
           if (context.getType<GqlContextType>() !== 'graphql') {
               throw new Error('OwnsCharacter è supportato solo sui resolver GraphQL');
           }
           const identity = readRequestIdentity(context);
           if (!identity) throw AuthException.unauthenticated();
           const gql = GqlExecutionContext.create(context);
           const targetId = rule.selectId(gql.getArgs());
           const access = await this.policy.authorize(identity, rule.resource, targetId);
           if (access === ResourceAccess.onBehalf) {
               this.decisions.recordOnBehalf(gql.getInfo<GraphQLResolveInfo>(), { type: rule.resource, id: targetId });
           }
           return true;
       }
   }
   ```
