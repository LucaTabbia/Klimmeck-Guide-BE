```typescript
export interface Distribution { count: number; min: number; median: number; max: number }
export interface CatalogStats {
  mode: 'stats';
  equipmentPowerByRarity: Record<string, Distribution>;   // somma dei power di damages.base + damages.energy per pezzo
  equipmentPowerByEquipType: Record<string, Distribution>;
  spellPowerByUseType: Record<string, Distribution>;      // energyDamage.power
  spellPower: Distribution;
  lootPowerByEffect: Record<string, Distribution>;
  hungerMarchMinutes: Distribution;                        // power × food.minutesPerPower
  winChanceExamples: { encounter: string; lifePoints: number; party: string; partyPower: number; winChance: number }[];
}
export function catalogStats(tree: ContentTree): CatalogStats;
// winChanceExamples (formula del resolver segnaposto: partyPower / (partyPower + lifePoints), 2 decimali):
//   Lupo 1100 vs «rookie, 1 pezzo mediano common»; Troll 8400 vs «paladin, 3 pezzi mediani rare»;
//   Drago rosso adulto 25600 vs «hero, 9 pezzi mediani legendary»; Tarrasque 67600 vs «5 legend × 9 pezzi mediani legendary»
// Mediana: elemento centrale dei valori ordinati (media dei due centrali se pari), arrotondata all'intero
```
