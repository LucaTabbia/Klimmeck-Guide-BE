export class SeededChanceSource extends ChanceSource {
  constructor(seed: number) // stato = seed >>> 0
  next(): number {          // [0, 1)
    let t = (this.state = (this.state + 0x6d2b79f5) >>> 0);
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  }
  pick<T>(items: readonly T[]): T | undefined;      // items[Math.floor(next() × length)]
  shuffle<T>(items: readonly T[]): T[];             // Fisher–Yates con next()
}
