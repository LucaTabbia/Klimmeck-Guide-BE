export const CONTENT_COLLECTIONS = ['lore', 'spells', 'equipmentitems', 'lootitems', 'enemies', 'pets', 'quests'] as const; // ordine di caricamento D-27
export type ContentCollection = (typeof CONTENT_COLLECTIONS)[number];
export const COLLECTION_DIRS: Record<ContentCollection, string> = { lore: 'lore', spells: 'spells', equipmentitems: 'equipment', lootitems: 'loot', enemies: 'enemies', pets: 'pets', quests: 'quests' };
export interface WorldPoi { id: string; type: string[]; location: [number, number] }
export interface WorldCity { id: string; name: string; type: string; markerLocation: string; relatedLore: string | null; pointsOfInterest: WorldPoi[] }
export interface ImageLink { id: string; url: string }
export interface ImagesFile { lore: ImageLink[]; enemies: ImageLink[]; pets: ImageLink[] }
export interface ContentTree {
  dir: string;
  entries: Record<ContentCollection, LoadedEntry[]>;   // tutti i *.yaml della cartella della collezione, file in ordine alfabetico
  fixes: LoadedEntry[];                                // content/fixes/*.yaml
  images: ImagesFile;                                  // content/images.yaml (assente → liste vuote)
  world: WorldCity[] | null;                           // content/world/cities.yaml (assente → null)
  config: ContentConfig | null;                        // content/config.yaml (assente o invalido → null + issue)
  issues: ContentIssue[];                              // errori di parsing, id, config
}
export function loadContentTree(dir: string): ContentTree; // sincrona (readdirSync/readFileSync), mai rete né env

// src/content/rules/content-config.ts
export type ContentTier = 'rookie' | 'adventurer' | 'paladin' | 'mage' | 'hero' | 'legend';
export type FixedQuestType = 'hunt' | 'dungeon' | 'heal' | 'job' | 'study' | 'crime' | 'guard';
export interface ContentConfig {
  enemy: { xpScale: number; hpScale: number; damageScale: number };
  tiers: Record<Exclude<ContentTier, 'legend'>, [number, number]>;
  mediumPeByLevel: number[];                       // 20 valori, colonna «Medio» del documento
  tierMultiplier: Record<Exclude<ContentTier, 'legend'>, number>;
  prizes: Record<FixedQuestType, { copper: number; xp: number; randomLootQuantity?: number }>;
  dungeonLootRarity: Record<Exclude<ContentTier, 'legend'>, string>;
  timeToCompleteHours: Record<FixedQuestType, number>;
  minTitle: { serviceCity: ContentTier; serviceCapital: ContentTier };
  dungeonTierByCitySize: { Village: ContentTier; City: ContentTier; SeaCity: ContentTier; Capital: ContentTier };
  dungeonHeroZones: string[];
  hunt: { maxAnimalPe: number; maxShareOfRegion: number };
  generator: { seed: number; crimeCityChance: number; studyCityChance: number };
  food: { minutesPerPower: number };
}
export function parseContentConfig(raw: unknown, file: string): { config: ContentConfig | null; issues: ContentIssue[] };
