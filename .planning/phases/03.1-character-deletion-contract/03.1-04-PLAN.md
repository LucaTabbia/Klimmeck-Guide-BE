async onModuleInit() {
    const pipeline = [{ $match: { operationType: { $in: ['update', 'replace'] } } }];
    this.changeStream = this.characterModel.watch(pipeline, { fullDocument: 'updateLookup' });
    this.changeStream.on('change', async (change) => {
        try {
            if (change.fullDocument) {
                const id = change.fullDocument._id;
                const character = await this.findOne(id)
                this.pubSub.publish('characterUpdated', { characterUpdated: character });
            }
        } catch (err) {
            this.logger.error('Errore nel change stream', err);
        }
    });
    this.changeStream.on('error', (err) => { this.logger.error('Errore nel change stream MongoDB', err); });
}
