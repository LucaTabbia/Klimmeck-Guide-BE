           }

       }

           const catalog = await this.catalog();

           const repaired = await this.characterModel.updateOne(filter, spellStatesRepairPipeline(catalog)).exec();
           const trimFilter = { ...outOfRangeFilter(catalog), _id: characterId };
           await this.warnOutOfRange(trimFilter, catalog);
           const trimmed = await this.characterModel.updateOne(trimFilter, spellStatesTrimPipeline(catalog)).exec();
           return repaired.modifiedCount === 1 || trimmed.modifiedCount === 1;

   @Injectable()
   export class SpellRecoveryService {
       private readonly logger = new Logger(SpellRecoveryService.name);
       constructor(
           @InjectModel(Character.name) private readonly characterModel: Model<CharacterDocument>,
           @InjectModel(Spell.name) private readonly spellModel: Model<SpellDocument>,
           private readonly migrator: SpellStateMigrator,
           private readonly clock: Clock,
       ) {}

       // ogni istante scaduto torna come uso, uno per update, al massimo maxUsages volte; idempotente sotto replay e gara con lo sweep (D-06, D-09)
       async recoverDue(characterId: Types.ObjectId, spellId: Types.ObjectId): Promise<number> {
           const maxUsages = await this.maxUsagesOf(characterId, spellId);
           if (maxUsages === null) return 0;
           await this.migrator.migrateOne(characterId);
           let credited = 0;
           while (credited < maxUsages) {
               const credit = await this.characterModel
                   .updateOne(dueCreditFilter(characterId, spellId, maxUsages, this.clock.now()), CREDIT_DUE_USAGE, spellStateArrayFilters(spellId))
                   .exec();
               if (credit.modifiedCount !== 1) break;
               credited++;
           }
           if (credited > 0) this.logger.log(`Magia ${spellId.toHexString()} del personaggio ${characterId.toHexString()}: ${credited} usi restituiti`);
           return credited;
       }

       // job accodati prima del deploy: solo il deficit non tracciato (D-06 rivista)
       async recoverLegacy(characterId: Types.ObjectId, spellId: Types.ObjectId): Promise<boolean> {
           const maxUsages = await this.maxUsagesOf(characterId, spellId);
           if (maxUsages === null) return false;
           await this.migrator.migrateOne(characterId);
           const credit = await this.characterModel
               .updateOne(legacyCreditFilter(characterId, spellId, maxUsages), CREDIT_LEGACY_USAGE, spellStateArrayFilters(spellId))
               .exec();
           return credit.modifiedCount === 1;
       }

       private async maxUsagesOf(characterId: Types.ObjectId, spellId: Types.ObjectId): Promise<number | null> {
           const spell = await this.spellModel.findById(spellId, { maxUsages: 1 }).lean<{ maxUsages?: unknown } | null>().exec();
           if (spell && Number.isSafeInteger(spell.maxUsages) && (spell.maxUsages as number) >= 0) return spell.maxUsages as number;
           this.logger.warn(`Magia ${spellId.toHexString()} assente dal catalogo: nessun recupero per il personaggio ${characterId.toHexString()}`);
           return null;
       }
   }
