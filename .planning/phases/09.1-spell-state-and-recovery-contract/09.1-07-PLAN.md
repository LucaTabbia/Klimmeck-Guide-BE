   ```typescript
   // sul filo (D-04a, D-05): usi dalla voce di stato, ordine degli slot, voci orfane scartate come status.spells
   export function spellsOnTheWire(
       status: { spells: Spell[]; spellStates: SpellState[] },
       active: ActiveSpell[],
   ): { spellStates: SpellState[]; activeSpells: ActiveSpell[] } {
       const known = new Map(status.spells.filter((spell) => spell?.id).map((spell) => [String(spell.id), spell]));
       const spellStates = (status.spellStates ?? []).filter((state) => {
           const spell = state.spell as Spell | null;
           return Boolean(spell && spell.id && known.has(String(spell.id)));
       });
       const usagesOf = new Map(spellStates.map((state) => [String((state.spell as Spell).id), state.usages]));
       const activeSpells = (active ?? []).flatMap((entry) => {
           const id = refIdOf(entry.spell)?.toHexString() ?? (entry.spell as Spell | null)?.id;
           const spell = id ? known.get(String(id)) : undefined;
           return spell ? [{ spell, usages: usagesOf.get(String(id)) ?? entry.usages }] : [];
       });
       return { spellStates, activeSpells };
   }
   ```
