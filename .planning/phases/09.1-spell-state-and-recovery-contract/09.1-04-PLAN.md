   ```typescript
   interface DueSpellStates { _id: Types.ObjectId; status: { spellStates?: { spell: Types.ObjectId; recoveries?: Date[] }[] } }

   @Injectable()
   export class SpellRecoverySweeper implements OnApplicationBootstrap, OnModuleDestroy {
       private readonly logger = new Logger(SpellRecoverySweeper.name);
       private timer: NodeJS.Timeout | null = null;
       private inFlight: Promise<number> | null = null;
       private stopping = false;
       constructor(
           @Inject(SPELL_RECOVERY_CONFIG) private readonly config: SpellRecoveryConfig,
           @InjectModel(Character.name) private readonly characterModel: Model<CharacterDocument>,
           private readonly recovery: SpellRecoveryService,
           private readonly clock: Clock,
       ) {}

       // il DB decide (D-09): reconciler sempre al boot, mai atteso; 0 spegne solo lo sweep periodico
       onApplicationBootstrap(): void { /* come TravelSweeper */ }
       async onModuleDestroy(): Promise<void> { /* stopping = true, clearInterval, await inFlight?.catch */ }
       sweep(): Promise<number> { /* inFlight ??= runOnce().finally(...) */ }

       private async runOnce(): Promise<number> {
           const now = this.clock.now();
           const due = await this.characterModel
               .find({ ...ACTIVE_CHARACTER_FILTER, 'status.spellStates.recoveries': { $lte: now } }, { 'status.spellStates': 1 })
               .lean<DueSpellStates[]>()
               .exec();
           let credited = 0;
           // in serie: nessun picco al boot; ogni voce è un update condizionale indipendente
           for (const character of due) {
               for (const state of character.status.spellStates ?? []) {
                   if (this.stopping) return credited;
                   const first = state.recoveries?.[0];
                   if (!first || first.getTime() > now.getTime()) continue;
                   try {
                       credited += await this.recovery.recoverDue(character._id, state.spell);
                   } catch (error) {
                       this.logger.error(`Recupero della magia ${state.spell.toHexString()} del personaggio ${character._id.toHexString()} fallito: ${describeError(error)}`);
                   }
               }
           }
           if (credited > 0) this.logger.log(`Sweep dei recuperi delle magie: ${credited} usi restituiti`);
           return credited;
       }
   }
   ```
