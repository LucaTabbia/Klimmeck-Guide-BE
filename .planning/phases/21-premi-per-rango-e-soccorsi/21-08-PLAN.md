export enum DungeonTurnKind { exploration = 'exploration', fight = 'fight' }

export interface DungeonTurn { index: number; kind: DungeonTurnKind; enemyIndex: number }

export interface RescueTurn { index: number; kind: DungeonTurnKind; total: number }

export interface TurnCounters { turnsTotal: number; turnsCompleted: number; lastTurn: number; lastTurnKind: DungeonTurnKind }
export function turnCountersOf(resolution: Pick<DungeonResolution, 'verdict' | 'turnsTotal' | 'turnsCompleted' | 'failedTurn'>): TurnCounters;
// successo: { turnsTotal, turnsCompleted, lastTurn: turnsCompleted, lastTurnKind: tipo del turno turnsCompleted − 1 }
// sconfitta: { turnsTotal, turnsCompleted: failedTurn.index, lastTurn: failedTurn.index + 1, lastTurnKind: failedTurn.kind }

    turnsTotal: number; turnsCompleted: number;       // turnsTotal = turni dell'intera dungeon

}
