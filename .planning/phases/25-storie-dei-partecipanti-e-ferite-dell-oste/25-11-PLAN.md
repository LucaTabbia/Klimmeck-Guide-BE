export enum MissionDraftStep { teleport, fights, prizes, recap, stories }   // stories = 'stories': ultima pagina, dopo il riepilogo (Phase 25 D-05)

// aggiunta facoltativa al background di un partecipante (Phase 25 D-06): testo già ripulito, mai vuoto
@Schema({ _id: false })
export class MissionBackgroundAdditionRecord {
    @Prop({ type: MongooseSchema.Types.ObjectId, ref: Character.name, required: true })
    character: Types.ObjectId;
    @Prop({ type: String, required: true })
    text: string;
}
// in MissionDraft
@Prop({ type: [MissionBackgroundAdditionRecordSchema], default: [] })
backgroundAdditions: MissionBackgroundAdditionRecord[];

// mission-draft.inputs.ts
@InputType()
export class MissionBackgroundAdditionInput {
    @Field(() => ID) @IsObjectIdHex() characterId: string;
    // forma sul filo: il tetto di dominio (1000 grafemi) risponde col suo codice (Phase 25 D-07); il testo vuoto è ammesso: nessun @IsNotEmpty né @MinLength (D-06)

}
// in SaveMissionDraftInput — sezione presente = sostituisce l'intera lista; testo vuoto = nessuna voce (Phase 25 D-06)
@Field(() => [MissionBackgroundAdditionInput], { nullable: true })
@IsOptional() @IsArray() @ArrayMaxSize(MAX_OVERRIDE_ENTRIES)
@ValidateNested({ each: true }) @Type(() => MissionBackgroundAdditionInput)
backgroundAdditions?: MissionBackgroundAdditionInput[] | null;

// mission-draft.wire.ts
@ObjectType()
export class MissionBackgroundAddition {
    @Field(() => ID) character: string;
    @Field(() => String) text: string;
}
// in MissionDraftView
@Field(() => [MissionBackgroundAddition])
backgroundAdditions: MissionBackgroundAddition[];
