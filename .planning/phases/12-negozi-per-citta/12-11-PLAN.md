```typescript
// ordine della lista a DB (la cura decide), id curati come stringa tollerati; serviti solo gli ammessi da D-05, senza ripetizioni
export function shelfInPoolOrder<T extends ShelfFlags & { _id: Types.ObjectId }>(
    poolIds: readonly unknown[] | null | undefined,
    items: readonly T[],
): T[] {
    const byId = new Map(items.map((item) => [item._id.toHexString(), item]));
    const served = new Set<string>();
    return (poolIds ?? []).flatMap((raw) => {
        const hex = refIdOf(raw)?.toHexString();
        const item = hex === undefined ? undefined : byId.get(hex);
        if (!item || served.has(hex!) || !isShelfEligible(item)) return [];
        served.add(hex!);
        return [item];
    });
}
```
