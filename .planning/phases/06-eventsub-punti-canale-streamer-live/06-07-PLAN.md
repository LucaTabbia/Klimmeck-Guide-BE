Controller di riferimento (RESEARCH Code Example §2):
```typescript
@Controller('webhooks/twitch')
export class EventSubController {
    constructor(private readonly webhook: EventSubWebhookService) {}
    @Public()
    @Post('eventsub')
    async receive(@Req() req: RawBodyRequest<Request>, @Res() res: Response): Promise<void> {
        const reply = await this.webhook.handle(req.headers, req.rawBody, req.body);
        switch (reply.kind) {
            case 'challenge': res.status(HttpStatus.OK).type('text/plain').send(reply.challenge); return;
            case 'noContent': res.status(HttpStatus.NO_CONTENT).end(); return;
            case 'forbidden': res.status(HttpStatus.FORBIDDEN).end(); return;
            case 'retryLater': res.status(HttpStatus.INTERNAL_SERVER_ERROR).end();
        }
    }
}
```
